       ```typescript
         db: BetterSQLite3Database<typeof schema>;

       ```

    2. REMOVE `fastify.decorate('db', db)` and its type declaration. Replace with:
       ```typescript
       declare module 'fastify' {
         interface FastifyRequest {
           getUserDb: () => BetterSQLite3Database<typeof schema>;
           getUserQdrant: () => TenantQdrantClient;
         }
       }
       ```

    3. Add dependency on 'registry' and 'qdrant' plugins (in fp() dependencies array). The plugin needs registry for events and qdrant for createTenantQdrant.

    4. On plugin init, create DBs for all existing users:
       ```typescript
       for (const user of fastify.registry.getAllUsers()) {
         createUserDb(user.userId);
       }
       ```

    5. Listen for registry events:
       - `user-added`: mkdir {dataDir}/{userId}, createDatabase({dataDir}/{userId}/index.db), add to Map
       - `user-removed`: close sqlite handle, delete from Map, rm -rf {dataDir}/{userId} directory, call fastify.purgeUserVectors(userId) to clean Qdrant

    6. Add onRequest hook (registered AFTER auth via plugin dependency order):
       ```typescript
       fastify.decorateRequest('getUserDb', null);
       fastify.decorateRequest('getUserQdrant', null);

       fastify.addHook('onRequest', async (request) => {
         if (!request.user) return; // unauthenticated routes (health, etc.)
         const userId = request.user.userId;

         request.getUserDb = () => {
           const entry = userDbs.get(userId);
           if (!entry) throw new Error(`No database for user: ${userId}`);
           return entry.db;
         };

         request.getUserQdrant = () => {
           return fastify.createTenantQdrant(userId);
         };
       });
       ```

    7. Delete old root-level index.db on startup:
       ```typescript
       const legacyDbPath = join(dataDir, 'index.db');
       try { await unlink(legacyDbPath); fastify.log.info('Deleted legacy root index.db'); }
       catch (e) { if ((e as NodeJS.ErrnoException).code !== 'ENOENT') throw e; }
       ```
       Also delete WAL/SHM files: index.db-wal, index.db-shm (same pattern)

    8. On close hook: iterate Map, close all sqlite handles.

    Rewrite src/plugins/__tests__/db.test.ts:
    - Test that DBs are created for all users from registry on init
    - Test user-added event creates new DB at correct path
    - Test user-removed event closes DB, deletes directory, calls purgeUserVectors
    - Test getUserDb returns correct DB for authenticated request

    - Test legacy index.db is deleted on startup

    - Use tmp directories for data dir (clean up in afterEach)
