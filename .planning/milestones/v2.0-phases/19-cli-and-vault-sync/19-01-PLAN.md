export const userRecordSchema = z.object({
  userId: z.string().min(1).regex(/^[a-z0-9-]+$/),
  apiKey: z.string().regex(/^cv-/),
  vaultPath: z.string().min(1),
  openaiKey: z.string().min(1),
  obsidian: z.object({
    email: z.string().email(),
    password: z.string().min(1),
    vault: z.string().min(1),
    token: z.string().optional(),
  }),
});

export type UserRecord = z.infer<typeof userRecordSchema>;

export class UserRegistry extends EventEmitter<RegistryEvents> {
  constructor(opts: UserRegistryOptions);  // { filePath, logger?, onReload?, onUserCountChange? }
  async load(): Promise<void>;
  getAllUsers(): UserRecord[];
  lookupByApiKey(apiKey: string): UserRecord | undefined;
  lookupByUserId(userId: string): UserRecord | undefined;
  async addUser(record: UserRecord): Promise<void>;
  async removeUser(userId: string): Promise<void>;
  static generateApiKey(): string;
}

    3. Create `src/cli/commands/add-user.ts` exporting `registerAddUser(program: Command)`:
       - Subcommand: `add-user` with `<name>` argument
       - Required options: `--obsidian-email`, `--obsidian-password`, `--vault`, `--openai-key`

         e. Create UserRegistry instance with filePath `${dataDir}/users.json`, call `load()`, then `addUser(record)`
         f. Print success message with userId and apiKey
       - On any error: print error message to stderr, call `process.exit(1)`
