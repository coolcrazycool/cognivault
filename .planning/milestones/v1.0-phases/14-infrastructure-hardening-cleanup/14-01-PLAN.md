```typescript
it('closes database connection on app.close() without error', async () => {
  const closeDir = await fs.mkdtemp(path.join(os.tmpdir(), 'db-close-test-'));
  const closeVault = path.join(closeDir, 'vault');
  const closeData = path.join(closeDir, 'data');
  await fs.mkdir(closeVault, { recursive: true });

  const origDataDir = process.env.COGNIVAULT_DATA_DIR;
  const origVaultPath = process.env.VAULT_PATH;
  process.env.COGNIVAULT_DATA_DIR = closeData;
  process.env.VAULT_PATH = closeVault;

  try {
    // vi.resetModules() is needed to get a fresh buildApp with new env
    vi.resetModules();
    const { buildApp: buildCloseApp } = await import('../../app.js');
    const closeApp = await buildCloseApp({ logger: false });
    await closeApp.ready();
    // Verify the app's db works before closing
    expect(closeApp.db).toBeDefined();
    // Close must resolve without throwing
    await expect(closeApp.close()).resolves.toBeUndefined();
  } finally {
    process.env.COGNIVAULT_DATA_DIR = origDataDir;
    process.env.VAULT_PATH = origVaultPath;
    await fs.rm(closeDir, { recursive: true, force: true });
  }
});
```
