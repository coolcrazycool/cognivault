export interface VaultEntry {
  name: string;
  path: string;       // relative to vault root
  type: 'file' | 'directory';
}

export interface ListOptions {
  path?: string;
  recursive?: boolean;
  ext?: string;
}

export interface ContentResult {
  path: string;
  content: string;
}

export class VaultManager {
  constructor(rootPath: string);
  initialize(): Promise<void>;
  resolvePath(relativePath: string): Promise<string>;
  listFiles(options: ListOptions): Promise<VaultEntry[]>;
  readContent(filePath: string): Promise<ContentResult>;
  readMetadata(filePath: string): Promise<MetadataResult>;
}

export class VaultError extends Error {
  readonly code: string;
  readonly statusCode: number;
}
export class PathTraversalError extends VaultError {}
export class FileNotFoundError extends VaultError {}
export class UnsupportedMediaTypeError extends VaultError {}

  <name>Task 1: Implement listFiles and readContent in VaultManager</name>
  <files>src/lib/vault.ts, src/lib/__tests__/vault.test.ts</files>
  <behavior>
    - listFiles({}) returns entries at vault root (files and directories)
    - listFiles({ path: 'subfolder' }) returns entries in that subfolder
    - listFiles({ recursive: true }) returns entries from all nested subdirectories
    - listFiles({ ext: 'md' }) returns only .md files
    - listFiles({ ext: 'md', recursive: true }) combines both filters
    - listFiles excludes dotfiles/dotfolders from results
    - listFiles excludes symlinks from results
    - listFiles on nonexistent directory throws FileNotFoundError
    - readContent('note.md') returns { path: 'note.md', content: 'markdown body' } with frontmatter stripped
    - readContent('no-frontmatter.md') returns full content as-is
    - readContent('image.png') throws UnsupportedMediaTypeError
    - readContent('nonexistent.md') throws FileNotFoundError
    - readContent on a file with UTF-8 Cyrillic content returns content correctly
  </behavior>
  <action>
    Update `src/lib/vault.ts` to implement the stubbed `listFiles` and `readContent` methods:

    **listFiles(options: ListOptions):**
    1. Resolve the target directory path via `resolvePath(options.path ?? '')`
    2. Verify it's a directory via `stat()`
    3. Use `readdir(dirPath, { withFileTypes: true, recursive: options.recursive ?? false })`
    4. For each entry: derive vault-relative path using `path.relative(rootPath, path.join(entry.parentPath ?? dirPath, entry.name))`
    5. Skip entries where any path segment starts with '.' (dotfiles/dotfolders)
    6. Skip symlinks (`entry.isSymbolicLink()`)
    7. Apply extension filter if `options.ext` provided (normalize: prepend '.' if missing)
    8. Return array of `VaultEntry` objects sorted alphabetically by path

    **readContent(filePath: string):**
    1. Resolve via `resolvePath(filePath)`
    2. Check if text file using extension-based allowlist (`.md`, `.markdown`, `.txt`, `.json`, `.yaml`, `.yml`, `.css`, `.js`, `.ts`, `.html`, `.xml`, `.csv`, `.svg`, `.canvas`, `.excalidraw`). If not in allowlist, throw `UnsupportedMediaTypeError`
    3. Read file with `readFile(resolved, 'utf-8')`
    4. Parse with gray-matter: `import matter from 'gray-matter'`
    5. Return `{ path: filePath, content: parsed.content.trim() }`
    6. If gray-matter throws on non-YAML content (e.g., JSON files), return raw content without stripping

    Add corresponding unit tests to `src/lib/__tests__/vault.test.ts`. Extend the existing test fixture to include: markdown files with/without frontmatter, nested subdirectories, a binary file (e.g., fake .png), a Cyrillic-named file.
  </action>
